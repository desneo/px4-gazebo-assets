cmake_minimum_required(VERSION 3.22.1)
project(gz_zhongke_d80_controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz-plugin2 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc video tracking)
set(LOCAL_CORE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/gimbal_core)
set(CONTROL_SOURCES src/control.cc src/kinematics.cc src/truth_tracking.cc src/inertial_los.cc
  ${LOCAL_CORE}/src/kinematics.cc ${LOCAL_CORE}/src/tracker.cc
  ${LOCAL_CORE}/src/mini_json.cc)
add_library(gz-zhongke-d80-controller SHARED src/gimbal_controller.cc ${CONTROL_SOURCES})
target_include_directories(gz-zhongke-d80-controller PRIVATE include ${LOCAL_CORE}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(gz-zhongke-d80-controller PRIVATE gz-sim8::core gz-transport13::core gz-msgs10::core gz-plugin2::core ${OpenCV_LIBRARIES})
set_target_properties(gz-zhongke-d80-controller PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
include(GNUInstallDirs)
install(TARGETS gz-zhongke-d80-controller LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
include(CTest)
if(BUILD_TESTING)
 add_executable(zhongke-truth-selection-test tests/truth_selection_test.cc src/truth_tracking.cc src/kinematics.cc ${LOCAL_CORE}/src/kinematics.cc)
 target_include_directories(zhongke-truth-selection-test PRIVATE include ${LOCAL_CORE}/include ${OpenCV_INCLUDE_DIRS})
 target_link_libraries(zhongke-truth-selection-test PRIVATE ${OpenCV_LIBRARIES})
 add_test(NAME zhongke-truth-selection COMMAND zhongke-truth-selection-test)
 add_executable(zhongke-contract-test tests/d80_contract_test.cc ${CONTROL_SOURCES})
 target_include_directories(zhongke-contract-test PRIVATE include ${LOCAL_CORE}/include ${OpenCV_INCLUDE_DIRS})
 target_link_libraries(zhongke-contract-test PRIVATE ${OpenCV_LIBRARIES})
 add_test(NAME zhongke-contract COMMAND zhongke-contract-test)
 add_executable(zhongke-control-test tests/control_test.cc ${CONTROL_SOURCES})
 target_include_directories(zhongke-control-test PRIVATE include ${LOCAL_CORE}/include ${OpenCV_INCLUDE_DIRS})
 target_link_libraries(zhongke-control-test PRIVATE ${OpenCV_LIBRARIES})
 add_test(NAME zhongke-control COMMAND zhongke-control-test)
 add_executable(zhongke-telemetry-test tests/telemetry_test.cc ${LOCAL_CORE}/src/mini_json.cc)
 target_include_directories(zhongke-telemetry-test PRIVATE include ${LOCAL_CORE}/include)
 add_test(NAME zhongke-telemetry COMMAND zhongke-telemetry-test)
 add_executable(zhongke-control-schedule-test tests/control_schedule_test.cc)
 target_include_directories(zhongke-control-schedule-test PRIVATE include)
 add_test(NAME zhongke-control-schedule COMMAND zhongke-control-schedule-test)
 add_executable(zhongke-target-los-test tests/inertial_los_test.cc src/inertial_los.cc ${LOCAL_CORE}/src/kinematics.cc)
 target_include_directories(zhongke-target-los-test PRIVATE include ${LOCAL_CORE}/include)
 add_test(NAME zhongke-target-los COMMAND zhongke-target-los-test)
endif()
