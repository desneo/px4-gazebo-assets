cmake_minimum_required(VERSION 3.22.1)
project(gz_two_axis_pod_controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz-plugin2 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc video tracking)
set(SHARED_CONTROLLER ${CMAKE_CURRENT_SOURCE_DIR}/../../three_axis_pod/controller)
set(CONTROL_SOURCES src/control.cc src/kinematics.cc
  ${SHARED_CONTROLLER}/src/kinematics.cc ${SHARED_CONTROLLER}/src/tracker.cc
  ${SHARED_CONTROLLER}/src/mini_json.cc)
add_library(gz-two-axis-pod-controller SHARED src/gimbal_controller.cc ${CONTROL_SOURCES})
target_include_directories(gz-two-axis-pod-controller PRIVATE include ${SHARED_CONTROLLER}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(gz-two-axis-pod-controller PRIVATE gz-sim8::core gz-transport13::core gz-msgs10::core gz-plugin2::core ${OpenCV_LIBRARIES})
set_target_properties(gz-two-axis-pod-controller PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
include(GNUInstallDirs)
install(TARGETS gz-two-axis-pod-controller LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
include(CTest)
if(BUILD_TESTING)
 add_executable(two-axis-control-test tests/control_test.cc ${CONTROL_SOURCES})
 target_include_directories(two-axis-control-test PRIVATE include ${SHARED_CONTROLLER}/include ${OpenCV_INCLUDE_DIRS})
 target_link_libraries(two-axis-control-test PRIVATE ${OpenCV_LIBRARIES})
 add_test(NAME two-axis-control COMMAND two-axis-control-test)
endif()
