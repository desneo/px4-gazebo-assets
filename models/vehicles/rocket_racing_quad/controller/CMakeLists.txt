cmake_minimum_required(VERSION 3.22)
project(rocket_flight_vibration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz-plugin2 REQUIRED)
add_library(gz-rocket-vibration SHARED src/flight_vibration.cc)
target_include_directories(gz-rocket-vibration PRIVATE include)
target_link_libraries(gz-rocket-vibration PRIVATE gz-sim8::core gz-transport13::core gz-msgs10::core gz-plugin2::core)
set_target_properties(gz-rocket-vibration PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
include(CTest)
if(BUILD_TESTING)
  add_executable(vibration-signal-test tests/signal_test.cc)
  target_include_directories(vibration-signal-test PRIVATE include)
  add_test(NAME vibration-signal COMMAND vibration-signal-test)
endif()
